@search_account_with_bank_and_currency_template = {
    rrel_init_template:{
        rrel_template:[*
            @input_param1=(.._bank <-_ concept_bank);;
            @input_param2=(.._currency <-_ concept_currency);;
            nrel_belongs_to_bank _-> (.._account _=> .._bank);;
            nrel_has_currency _-> (.._account _=> .._currency);;
            @output_param1=(.._account <-_ concept_bank_account);;
        *];
        rrel_template_input_params:{
            @input_param1;
            @input_param2
        };
        rrel_template_output_params:{
            @output_param1
        }
    }(*<- nrel_search_set_template;;*)
};;

            @output_param2=(.._parent _=> .._concept);;
            @output_param3=(.._concept _-> .._instance);;
            @output_param4=(.._relation _=> .._concept);;

            @input_param1
        };

            @output_param1;
